## 1). Load CSV Files

#converts the two CSV files (counts.csv and meta_data.csv) to data frames
#returns the two data frames

count_matrix_df <- read.csv("counts.csv")
meta_data_df <- read.csv("meta_data.csv")

## ------------------------------------------------------------

## 2). Generate Summary Statistics
#Using the count data from the selected gene, compute and report summary statistics, such as mean, median, standard deviation, #minimum, and maximum.

library(knitr)
library(dplyr)

#checks first column of count_matrix_df
#returns row in count_matrix_df where the gene is ENSG00000012048.23 for BRCA1
#returns row in count_matrix_df where the gene is ENSG00000139618.16 for BRCA2
#uses as.numeric to makes sure all summary stats returned are numerics

gene_row1 <- as.numeric(count_matrix_df[count_matrix_df[, 1] == 
                                         "ENSG00000012048.23", -1])

gene_row2 <- as.numeric(count_matrix_df[count_matrix_df[, 1] == 
                                          "ENSG00000139618.16", -1])

brca1_stats <- data.frame( #create a data frame with summary statistics for BRCA1
  Gene = "BRCA1",
  Min = min(gene_row1, na.rm = TRUE),
  Mean = mean(gene_row1, na.rm = TRUE),
  Median = median(gene_row1, na.rm = TRUE),
  Max = max(gene_row1, na.rm = TRUE),
  Sd = sd(gene_row1, na.rm = TRUE)
)

brca2_stats <- data.frame( #create a data frame with summary statistics for BRCA2
  Gene = "BRCA2",
  Min = min(gene_row2, na.rm = TRUE),
  Mean = mean(gene_row2, na.rm = TRUE),
  Median = median(gene_row2, na.rm = TRUE),
  Max = max(gene_row2, na.rm = TRUE),
  Sd = sd(gene_row2, na.rm = TRUE)
)

#combine BRCA1 and BRCA2 data frames into combined_stats data frame
combined_stats <- bind_rows(brca1_stats, brca2_stats)

#use knitr to convert combined_stats data frame into a table format
knitr::kable(combined_stats, 
             caption = "Summary Statistics", 
             align = "lccccc")

## 3). Create a Histogram
#Use ggplot2 to generate a histogram of the count data for the selected gene. This visualization should effectively display the distribution of the counts.

library(ggplot2)

#filter to only include row where ensembl gene ID = ENSG00000012048.23 (BRCA1 gene)
#store this row as a data frame called brca1_df
brca1_df <- as.data.frame(count_matrix_df[count_matrix_df[, 1] == "ENSG00000012048.23", -1])

#transpose brca1_df so that all expression values for BRCA1 are in a single column
#convert this transposed matrix into a data frame
brca1_df_transpose <- as.data.frame(t(brca1_df))
colnames(brca1_df_transpose) <- "Expression_Value_BRCA1" #column name

#create histogram showing frequency of expression values for BRCA1
brca1_hist <- ggplot(brca1_df_transpose, aes(x = Expression_Value_BRCA1)) +
  geom_histogram(binwidth = 250,
                 fill="#F2A2B1", 
                 color="#8B2252",
                 alpha = 0.8) +
  labs(
    x = "Expression Value",
    y = "Frequency",
    title = "Distribution of Expression Values For BRCA1 Gene"
  ) +
  theme_minimal() +
  theme(
    axis.text = element_text(size = 12, color = "black"),
    axis.title = element_text(size = 14, color = "black"),
    plot.title = element_text(size = 14, color = "black", hjust = 0.5)
  )

#save histogram as 5x7 pdf image
ggsave("histogram.pdf", plot = brca1_hist, width = 7, height = 5, units = "in")

print(brca1_hist)
